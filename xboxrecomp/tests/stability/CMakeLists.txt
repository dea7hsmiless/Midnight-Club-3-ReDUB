cmake_minimum_required(VERSION 3.20)
project(runtime_stability C)
set(CMAKE_C_STANDARD 11)
if(WIN32)
    # Match the game: the native D3D host omits SDL host/thread helpers.
    set(XBOXRECOMP_WIN_SDL_GL ON CACHE BOOL "" FORCE)
endif()
add_subdirectory(../.. runtime)
enable_testing()
foreach(test IN ITEMS kernel_bridge kernel_audio_setting)
    add_executable(${test}_test ../${test}/test_main.c)
    target_include_directories(${test}_test PRIVATE ../../src/kernel)
    target_link_libraries(${test}_test PRIVATE xboxrecomp)
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
add_executable(thread_suspend_test thread_suspend_test.c)
target_include_directories(thread_suspend_test PRIVATE ../../src/kernel)
target_link_libraries(thread_suspend_test PRIVATE xboxrecomp)
add_test(NAME thread_suspend COMMAND thread_suspend_test)
if(WIN32)
    add_executable(pushbuffer_test pushbuffer_test.c ../../src/kernel/nv2a_pb_scan.c)
    add_test(NAME pushbuffer COMMAND pushbuffer_test)
    foreach(test IN ITEMS kernel_directory kernel_object_paths kernel_irql_abi kernel_dispatch idex_channel)
        if(test STREQUAL "kernel_dispatch")
            set(source ../kernel_dispatch/kernel_dispatch_thread_test.c)
        elseif(test STREQUAL "idex_channel")
            set(source ../idex_channel/idex_channel_test.c)
        else()
            set(source ../${test}/test_main.c)
        endif()
        add_executable(${test}_test ${source})
        target_include_directories(${test}_test PRIVATE ../../src/kernel)
        target_link_libraries(${test}_test PRIVATE xboxrecomp)
        add_test(NAME ${test} COMMAND ${test}_test)
    endforeach()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_executable(file_concurrency_test file_concurrency_test.c)
    target_include_directories(file_concurrency_test PRIVATE ../../src/kernel ../../src/platform)
    target_link_libraries(file_concurrency_test PRIVATE xboxrecomp)
    target_link_options(file_concurrency_test PRIVATE -Wl,--wrap=lseek)
    add_test(NAME file_concurrency COMMAND file_concurrency_test)
    add_executable(kernel_memory_posix_test ../kernel_memory_posix/test_main.c)
    target_include_directories(kernel_memory_posix_test PRIVATE ../../src/kernel)
    target_link_libraries(kernel_memory_posix_test PRIVATE xboxrecomp)
    add_test(NAME kernel_memory_posix COMMAND kernel_memory_posix_test)
endif()
