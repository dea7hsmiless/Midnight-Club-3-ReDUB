# mcdub3remix_env.txt -- settings for Midnight Club 3: DUB Edition Remix
# (the Windows build). This is the reference copy; the game writes its own
# copy of it beside the executable on the first run and rewrites single keys
# there as you change them in game (Options > Port Options).
#
# One KEY=VALUE per line. '#' starts a comment. Read once at start, so restart
# the game after editing. Every setting below is commented out and shows the
# default: remove the '#' in front of a line to change it.
#
# The log's "settings:" line lists every setting actually in effect, which is
# the quickest way to check that a line took effect (it is written to
# mcdub3remix_log.txt beside the executable, or to the console if one is open).

# ============================================================================
# Disc and saves
# ============================================================================

# Where career saves live. Default: save\ beside the executable.

# ============================================================================

# 1.5 gives 1280x720; 2 is the published release value.
#RECOMP_GL_SCALE=2

# Menu and HUD art redrawn N times larger on the GPU (1 = off, up to 5).
#RECOMP_GL_UI_SCALE=5

# Anti-aliasing of edges (FXAA): 0 off, 1 FXAA, 2 FXAA low, 3 FXAA high.
#RECOMP_GL_FXAA=1

# Anisotropic filtering, 1 (off) to 16: sharp textures at shallow angles.
#RECOMP_GL_ANISO=16

# Sharpening strength, 0 (off) to 1.
#RECOMP_GL_SHARPEN=0.5

# Vertical sync: 0 off, 1 on, -1 adaptive.
#RECOMP_GL_VSYNC=0

# 30 is the Xbox's cadence and the default. 60 lifts the title's 30 Hz

#MC3_FRAME_RATE=30

# Windowed, borderless fullscreen, or exclusive fullscreen (restart).
#MC3_FULLSCREEN=0

# A fixed window size, or auto for the desktop's own.
#MC3_RESOLUTION=auto

# Tell the game the screen is widescreen: it renders 16:9 itself.

# One of the ready-made looks (0 custom, 1 performance, 2 balanced, 3 quality),

#MC3_GRAPHICS_PRESET=0

# Gamma (0.8 .. 1.4) and brightness in percent (80 .. 120).
#MC3_GAMMA=1.0

#MC3_FPS_OVERLAY=0

# Folder of replacement textures made with make hd-dump / make hd-upscale.

#MC3_ROAM_JOIN=203.0.113.10

# System Link (the console's LAN mode, which the online races use). On by
